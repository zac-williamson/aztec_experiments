use aztec::{
    hash::hash_args,
    protocol::{address::AztecAddress, abis::{gas::Gas, gas_fees::GasFees, gas_settings::GasSettings}, merkle_tree::root::compute_tree_root},
    test::helpers::{test_environment::TestEnvironment, authwit::add_private_authwit_from_call, txe_oracles},
};
use fee_common::{Policy, coupon_leaf};
use fee_target::FeeTarget;
use fee_sponsor::RestrictedSponsor;

// These tests exercise the actual TXE private call oracle with None as caller. Unlike
// env.call_private(from,...), this preserves the root entrypoint's real no-sender semantics.
unconstrained fn root_call(sponsor: AztecAddress, owner: AztecAddress, index: u32, blind: Field, sibling: Field, value: u32, nonce: Field, gas: GasSettings) {
    let call = RestrictedSponsor::at(sponsor).sponsor(owner,index,blind,sibling,value,nonce);
    let _result: [Field;0] = txe_oracles::private_call_new_flow(
        Option::none(), call.target_contract, call.selector, call.args, hash_args(call.args), false, [owner], [], gas,
    );
}
fn gas() -> GasSettings {
    // TXE constraints/composition only: zero prices avoid pretending that this fixture funds or proves fees.
    GasSettings::new(Gas::new(1000000,10000000),Gas::new(0,0),GasFees::new(0,0),GasFees::new(0,0))
}
unconstrained fn setup() -> (TestEnvironment, AztecAddress, AztecAddress, AztecAddress, AztecAddress, AztecAddress, Policy, Field, Field) {
    let mut env=TestEnvironment::new();
    let admin=env.create_light_account();
    let owner0=env.create_contract_account();
    let owner1=env.create_contract_account();
    let sponsor=env.deploy("@fee_sponsor/RestrictedSponsor").with_public_initializer(admin,RestrictedSponsor::interface().constructor(admin));
    let target=env.deploy("@fee_target/FeeTarget").with_public_initializer(admin,FeeTarget::interface().constructor(sponsor));
    let (chain,version)=env.utility_context(|context| (context.chain_id(),context.version()));
    let a=coupon_leaf(chain,version,sponsor,7,0,owner0,13);
    let b=coupon_leaf(chain,version,sponsor,7,1,owner1,15);
    let policy=Policy {root:compute_tree_root([a,b]),target,epoch:7,valid_from:0,valid_until:env.last_block_timestamp()+3600,
        max_da_gas:1000000,max_l2_gas:10000000,max_teardown_da:0,max_teardown_l2:0,
        max_fee_da:1,max_fee_l2:1,max_priority_da:0,max_priority_l2:0,max_fee_per_ticket:11000000,epoch_budget:22000000};
    env.call_public(admin,RestrictedSponsor::at(sponsor).configure(policy));
    (env,admin,owner0,owner1,sponsor,target,policy,a,b)
}
#[test]
unconstrained fn valid_two_owners_two_tickets_real_authwit() {
    let (env,_admin,owner0,owner1,sponsor,target,_policy,a,b)=setup();
    add_private_authwit_from_call(env,owner0,sponsor,FeeTarget::at(target).delegated(owner0,3,1));
    root_call(sponsor,owner0,0,13,b,3,1,gas());
    assert(env.view_public(FeeTarget::at(target).get_total())==3);
    add_private_authwit_from_call(env,owner1,sponsor,FeeTarget::at(target).delegated(owner1,5,2));
    root_call(sponsor,owner1,1,15,a,5,2,gas());
    assert(env.view_public(FeeTarget::at(target).get_total())==8);
}

}

#[test(should_fail_with="already present")]
unconstrained fn reject_ticket_replay_with_fresh_action_authwit() {
    let (env,_admin,owner0,_owner1,sponsor,target,_policy,_a,b)=setup();
    add_private_authwit_from_call(env,owner0,sponsor,FeeTarget::at(target).delegated(owner0,3,1));
    root_call(sponsor,owner0,0,13,b,3,1,gas());
    assert(env.view_public(FeeTarget::at(target).get_total())==3);
    // A new operation and authwit removes authwit replay as the intended rejection cause.
    add_private_authwit_from_call(env,owner0,sponsor,FeeTarget::at(target).delegated(owner0,5,2));
    root_call(sponsor,owner0,0,13,b,5,2,gas());
}
#[test(should_fail_with="already exists")]
unconstrained fn reject_policy_reinitialization() {
    let (env,admin,_owner0,_owner1,sponsor,_target,policy,_a,_b)=setup();
    let mut replacement=policy; replacement.epoch=8;
    env.call_public(admin,RestrictedSponsor::at(sponsor).configure(replacement));
}

unconstrained fn reject_actual_nested_private_call() {

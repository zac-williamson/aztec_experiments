import subprocess, time, json, pathlib, datetime
root = pathlib.Path.cwd()
directory = root / '.build/P04-crs-experiment'
started = time.monotonic()
results = []
for mode in ['derive', 'uncompressed']:
    command = ['/Users/zac/.nvm/versions/node/v24.15.0/bin/node', str(directory / 'run.mjs'), mode]
    with (directory / (mode + '.stdout')).open('w') as out, (directory / (mode + '.stderr')).open('w') as err:
        try:
            result = subprocess.run(command, cwd=root, stdout=out, stderr=err,
                timeout=max(1, 600 - (time.monotonic() - started)), check=False)
            entry = {'mode': mode, 'command': command, 'exitCode': result.returncode}
        except subprocess.TimeoutExpired:
            entry = {'mode': mode, 'command': command, 'timedOut': True}
    results.append(entry)
    print(json.dumps(entry), flush=True)
    if entry.get('exitCode') != 0:
        break
report = {'finishedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(),
    'infrastructureTimeoutSeconds': 600, 'elapsedSeconds': time.monotonic() - started,
    'freshChildProcessPerMode': True, 'results': results}
(directory / 'supervision.json').write_text(json.dumps(report, indent=2) + '\n')
raise SystemExit(0 if len(results) == 2 and all(r.get('exitCode') == 0 for r in results) else 1)

import assert from 'node:assert/strict';
import fs from 'node:fs';
import path from 'node:path';
import { createHash } from 'node:crypto';
import { gunzipSync } from 'node:zlib';
import { BarretenbergSync, BackendType } from '@aztec/bb.js';
import { ROOT, assertNodeVersion, assertAztecPackages } from '../../scripts/toolchain.mjs';

assertNodeVersion(); assertAztecPackages();
const mode = process.argv[2];
assert.ok(['derive', 'uncompressed'].includes(mode));
const destination = path.join(ROOT, '.build/P04-crs-experiment');
const sha = bytes => createHash('sha256').update(bytes).digest('hex');
const manifestBytes = fs.readFileSync(path.join(ROOT, 'crs-manifest.json'));
const manifest = JSON.parse(manifestBytes);
assert.equal(manifest.aztecVersion, '5.2.0');
const assets = Object.fromEntries(manifest.files.map(a => [a.name, a]));
assert.equal(assets['g1.dat'].numPoints, 1179648);
assert.equal(assets['grumpkin_g1.dat'].numPoints, 65537);
function verified(name) {
  const bytes = new Uint8Array(fs.readFileSync(path.join(ROOT, 'apps/dist/crs', name)));
  assert.equal(bytes.byteLength, assets[name].bytes);
  assert.equal(sha(bytes), assets[name].sha256);
  return bytes;
}
const wasmPath = path.join(ROOT, 'node_modules/@aztec/bb.js/dest/node/barretenberg_wasm/barretenberg-threads.wasm.gz');
const wasmGzip = fs.readFileSync(wasmPath);
const compressed = mode === 'derive' ? verified('g1.dat') : undefined;
const g2 = verified('g2.dat');
const grumpkin = verified('grumpkin_g1.dat');
let input = compressed;
if (mode === 'uncompressed') {
  const derivation = JSON.parse(fs.readFileSync(path.join(destination, 'derive.json')));
  assert.equal(derivation.outcome, 'pass');
  assert.equal(derivation.manifestSha256, sha(manifestBytes));
  assert.equal(derivation.wasmGzipSha256, sha(wasmGzip));
  input = new Uint8Array(fs.readFileSync(path.join(destination, 'g1-uncompressed.dat')));
  assert.equal(input.byteLength, 75497472);
  assert.equal(sha(input), derivation.output.sha256);
}
const record = { mode, startedAt: new Date().toISOString(), node: process.versions.node,
  platform: process.platform, arch: process.arch, packageVersion: JSON.parse(fs.readFileSync(path.join(ROOT, 'node_modules/@aztec/bb.js/package.json'))).version,
  backend: 'explicit BackendType.Wasm / fresh child process', proofGenerated: false,
  manifestSha256: sha(manifestBytes), wasmGzipSha256: sha(wasmGzip), wasmSha256: sha(gunzipSync(wasmGzip)),
  scriptSha256: sha(fs.readFileSync(new URL(import.meta.url))),
  input: { bytes: input.byteLength, sha256: sha(input), numPoints: 1179648 },
  g2: assets['g2.dat'], grumpkin: assets['grumpkin_g1.dat'], timingsMs: {} };
const stage = async (name, action) => {
  const start = performance.now();
  console.error(JSON.stringify({ mode, stage: name, state: 'started' }));
  const result = await action();
  record.timingsMs[name] = Math.round(performance.now() - start);
  console.error(JSON.stringify({ mode, stage: name, state: 'completed', durationMs: record.timingsMs[name] }));
  return result;
};
let bb;
try {
  bb = await stage('wasm-instance', () => BarretenbergSync.new({ backend: BackendType.Wasm, wasmPath }));
  const result = await stage('bn254-initialize', () => bb.srsInitSrs({ pointsBuf: input, numPoints: 1179648, g2Point: g2 }));
  assert.ok(result.pointsBuf instanceof Uint8Array);
  assert.equal(result.pointsBuf.byteLength, mode === 'derive' ? 75497472 : 0);
  record.returnedBytes = result.pointsBuf.byteLength;
  if (mode === 'derive') {
    record.output = { bytes: result.pointsBuf.byteLength, sha256: sha(result.pointsBuf) };
    fs.writeFileSync(path.join(destination, 'g1-uncompressed.dat'), result.pointsBuf, { flag: 'wx' });
  }
  const grumpkinResult = await stage('grumpkin-initialize', () => bb.srsInitGrumpkinSrs({ pointsBuf: grumpkin, numPoints: 65537 }));
  assert.equal(grumpkinResult.dummy, 0);
  record.outcome = 'pass';
} catch (error) {
  record.outcome = 'fail'; record.error = error.stack;
  process.exitCode = 1;
} finally {
  if (bb) await bb.destroy();
  record.finishedAt = new Date().toISOString();
  record.resourceUsage = process.resourceUsage();
  fs.writeFileSync(path.join(destination, mode + '.json'), JSON.stringify(record, null, 2) + '\n');
  console.log(JSON.stringify(record, null, 2));
}
